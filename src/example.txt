import { styled, VariantProps } from "react-native-motif"
import { Text } from "react-native"

const title = styled({
  base: {
    fontSize: 24,
    fontWeight: "bold"
  },
  variants: {
    color: {
      primary: {
        color: "red"
      },
      secondary: {
        color: "blue"
      }
    }
  },
  defaultVariants: {
    color: "primary"
  }
})

interface TitleProps extends VariantProps<typeof title> {
    children: string
}

export function Title({ color, children }: TitleProps) {
  return (
    <Text style={title({ color })}>
      {children}
    </Text>
  )
}
